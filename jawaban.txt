Soal 1 Membuat Database

create database myshop;
===================================================
Soal 2 Membuat Table di Dalam Database

users
create table users(
id INTEGER AUTO_INCREMET PRIMAY KEY,
name varchar (255),
email varchar (255),
password varchar (255)
)
===================================================
categories
create table categories(
id integer AUTO_INCREMENT PRIMARY key,
name varchar (255)
)
===================================================
items
create table items(
id integer AUTO_INCREMENT primary key,
name varchar(255),
description varchar(255),
price integer,
stock integer,
category_id integer
FOREIGN key(category_id) REFRENCES categories(id)
)	
===================================================

Soal 4 Mengambil Data dari Database
a. Mengambil data users
SELECT id,name,email FROM users;

b. Mengambil data items
- Buatlah sebuah query untuk mendapatkan data item pada table items yang memiliki harga di atas 1000000 (satu juta).
SELECT *FROM items WHERE price > 1000000;
===================================================
- Buat sebuah query untuk mengambil data item pada table items yang memiliki name serupa atau mirip (like) dengan kata kunci “uniklo”, “watch”, atau “sang” (pilih salah satu saja).
SELECT *FROM items WHERE name "%b50%";
===================================================
c. Menampilkan data items join dengan kategori
SELECT items.*, categories.name as categories FROM items INNER JOIN categories on items.category_id = category_id;
